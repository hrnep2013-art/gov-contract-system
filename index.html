    ADMIN DASHBOARD โหมดบันทึกในระบบ (พร้อมเชื่อมต่อ Google Sheets)

    ระบบติดตามและสรุปข้อมูลผู้ทำสัญญาจ้างพนักงานราชการ

    ตรวจสอบรายชื่อผู้เข้าระบบมาทำสัญญาจ้าง สรุปยอดรายสำนัก/กอง หน่วยงาน กลุ่มงาน และเชื่อมต่อ Google Sheets

    หมายเหตุ: ทุกครั้งที่มีผู้กด "บันทึกข้อมูลและสร้างเอกสารสัญญาจ้าง" ระบบจะแสดงใน Dashboard นี้ทันที และหากใส่ Web App URL ไว้ ระบบจะส่งข้อมูลไปบันทึกลง Google Sheets อัตโนมัติ

    วิธีเชื่อมระบบเข้ากับ Google Sheets (ใช้เวลา 1 นาที):
    1. สร้างไฟล์ Google Sheets ใหม่ ไปที่เมนู ส่วนขยาย (Extensions) → Apps Script
    2. วางโค้ดด้านล่างนี้ลงในไฟล์ Code.gs แล้วกดบันทึก
    3. กดปุ่ม การทำให้ใช้งานได้ (Deploy) → การทำให้ใช้งานได้รายการใหม่ (New deployment) เลือกประเภท เว็บแอป (Web app)
    4. ตั้งค่าผู้มีสิทธิ์เข้าถึงเป็น ทุกคน (Anyone) กด Deploy แล้วคัดลอก Web App URL มาวางในช่องด้านบน
    function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Contracts') || SpreadsheetApp.getActiveSpreadsheet().insertSheet('Contracts');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Timestamp', 'ContractNo', 'Prefix', 'FullName', 'NationalId', 'Bureau', 'SubUnit', 'EmpType', 'JobGroup', 'PositionTitle', 'SalaryAmount', 'DurationText', 'StartDate', 'EndDate', 'Phone', 'Email', 'RawJson']);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(), data.contractNo, data.prefix, data.fullName, data.nationalId,
    data.bureau, data.subUnit, data.empType, data.jobGroup, data.positionTitle,
    data.salaryAmount, data.durationText, data.startDateThai, data.endDateThai,
    data.phone, data.email, JSON.stringify(data)
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Contracts');
  if (!sheet || sheet.getLastRow() <= 1) {
    return ContentService.createTextOutput(JSON.stringify({ records: [] })).setMimeType(ContentService.MimeType.JSON);
  }
  var rows = sheet.getDataRange().getValues();
  var records = [];
  for (var i = 1; i < rows.length; i++) {
    try { records.push(JSON.parse(rows[i][16])); } catch (err) {}
  }
  return ContentService.createTextOutput(JSON.stringify({ records: records })).setMimeType(ContentService.MimeType.JSON);
}
